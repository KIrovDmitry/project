export interface ComparisonProfile {
  id: string;
  name: string;
  color: string;
  enabled: boolean;
  selection_type: 'formula' | 'manual';
  space_type: 'all' | 'cube' | 'sphere';
  distribution_type: 'all' | 'random' | 'gaussian' | 'grid' | 'fractal_tetrahedron' | 'fractal_tetrahedron_cropped' | 'fractal_octahedron' | 'fractal_octahedron_cropped' | 'fractal_menger' | 'fractal_cube';
  lambdaVal: 'all' | number;
  alphaVal: 'all' | number;
  model_type: 'all' | 'lpn' | 'er' | 'ba';
  er_pVal: 'all' | number;
  ba_mVal: 'all' | number;
  selectedNetworkIds: number[];
}

export const matchNetworkWithProfile = (net: Network, prof: ComparisonProfile): boolean => {
  if (prof.selection_type === 'manual') {
    return prof.selectedNetworkIds.includes(net.id);
  }

  if (prof.space_type !== 'all' && net.space_type !== prof.space_type) return false;
  if (prof.distribution_type !== 'all' && net.distribution_type !== prof.distribution_type) return false;

  const netModel = net.model_type || 'lpn';
  const profModel = prof.model_type || 'all';
  if (profModel !== 'all' && netModel !== profModel) return false;

  if (prof.lambdaVal !== 'all') {
    const l1 = net.lambda ? parseFloat(net.lambda.toFixed(5)) : 0;
    const l2 = typeof prof.lambdaVal === 'string' ? parseFloat(prof.lambdaVal) : prof.lambdaVal;
    if (Math.abs(l1 - l2) > 1e-5) return false;
  }

  if (prof.alphaVal !== 'all') {
    const a1 = net.alpha ? parseFloat(net.alpha.toFixed(5)) : 0;
    const a2 = typeof prof.alphaVal === 'string' ? parseFloat(prof.alphaVal) : prof.alphaVal;
    if (Math.abs(a1 - a2) > 1e-5) return false;
  }

  if (prof.er_pVal && prof.er_pVal !== 'all') {
    const p1 = net.er_p ?? 0;
    const p2 = typeof prof.er_pVal === 'string' ? parseFloat(prof.er_pVal) : prof.er_pVal;
    if (Math.abs(p1 - p2) > 1e-5) return false;
  }

  if (prof.ba_mVal && prof.ba_mVal !== 'all') {
    const m1 = net.ba_m ?? 0;
    const m2 = typeof prof.ba_mVal === 'string' ? parseInt(prof.ba_mVal) : prof.ba_mVal;
    if (m1 !== m2) return false;
  }

  return true;
};

import { useState, useEffect } from 'react';

export default function App() {
  const [activeTab, setActiveTab] = useState<'generator' | 'analyzer' | 'experiments'>('generator');
  const [networks, setNetworks] = useState<Network[]>([]);
  const [experiments, setExperiments] = useState<Experiment[]>([]);
  const [isLoading, setIsLoading] = useState<boolean>(false);

  const fetchData = async () => {
    try {
      const netRes = await fetch('/api/networks');
      if (netRes.ok) {
        const netData = await netRes.json();
        setNetworks(netData);
      }
      
      const expRes = await fetch('/api/experiments');
      if (expRes.ok) {
        const expData = await expRes.json();
        setExperiments(expData);
      }
    } catch (err) {
      console.error('Failed to connect to backend api:', err);
    }
  };

  useEffect(() => {
    fetchData();
  }, [activeTab]);

  const handleGenerateNetwork = async (formData: any) => {
    if (formData === null) {
      setIsLoading(true);
      await fetchData();
      setIsLoading(false);
      return null;
    }
    setIsLoading(true);
    try {
      const res = await fetch('/api/generate', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(formData),
      });
      
      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.detail || 'Failed to generate network');
      }
      
      await fetchData();
      setIsLoading(false);
      return data;
    } catch (err) {
      setIsLoading(false);
      throw err;
    }
  };

  const handleDeleteNetwork = async (id: number) => {
    try {
      const res = await fetch(`/api/networks/${id}`, {
        method: 'DELETE',
      });
      if (res.ok) {
        setNetworks((prev) => prev.filter((n) => n.id !== id));
      }
    } catch (err) {
      console.error('Failed to delete network:', err);
    }
  };

  const [analysisProgress, setAnalysisProgress] = useState<{ current: number; total: number; message: string } | null>(null);
  const [experimentProgress, setExperimentProgress] = useState<{ current: number; total: number; message: string } | null>(null);

  const readProgressStream = async (
    res: Response,
    onProgress: (progress: { current: number; total: number; message: string }) => void,
    onSuccess: (data: any) => void
  ) => {
    const reader = res.body?.getReader();
    const decoder = new TextDecoder('utf-8');
    let buffer = '';
    
    if (reader) {
      while (true) {
        const { value, done } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true });
        const lines = buffer.split('\n');
        buffer = lines.pop() || '';
        
        for (const line of lines) {
          if (line.trim()) {
            try {
              const step = JSON.parse(line);
              if (step.type === 'progress') {
                onProgress({
                  current: (step.index ?? 0) + 1,
                  total: step.total ?? 100,
                  message: step.message || '',
                });
              } else if (step.type === 'success') {
                onSuccess(step.results);
              }
            } catch (err) {
              console.error('Error parsing stream line:', err);
            }
          }
        }
      }
      if (buffer.trim()) {
        try {
          const step = JSON.parse(buffer);
          if (step.type === 'success') {
            onSuccess(step.results);
          }
        } catch (err) {}
      }
    }
  };

  const handleAnalyze = async (networkIds: number[], checkboxes: string[], threads: number) => {
    setIsLoading(true);
    setAnalysisProgress({ current: 0, total: networkIds.length, message: 'Инициализация анализа свойств...' });
    try {
      const res = await fetch('/api/analyze', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          network_ids: networkIds,
          checkboxes,
          threads,
        }),
      });
      
      if (!res.ok) {
        throw new Error(await res.text() || 'Failed to analyze');
      }

      await readProgressStream(
        res,
        (prog) => setAnalysisProgress(prog),
        (results) => {}
      );
      
      await fetchData();
    } catch (err) {
      console.error('Failed to analyze:', err);
    } finally {
      setIsLoading(false);
      setAnalysisProgress(null);
    }
  };

  const handleAnalyzeAllUnanalyzed = async (checkboxes: string[], threads: number) => {
    setIsLoading(true);
    setAnalysisProgress({ current: 0, total: 100, message: 'Подготовка общего анализа нерассчитанных свойств...' });
    try {
      const queryParams = checkboxes.map((col) => `checkboxes=${col}`).join('&');
      const res = await fetch(`/api/analyze-all-unanalyzed?${queryParams}&threads=${threads}`, {
        method: 'POST',
      });
      if (!res.ok) {
        throw new Error(await res.text() || 'Failed to analyze uncalculated');
      }

      await readProgressStream(
        res,
        (prog) => setAnalysisProgress(prog),
        (results) => {}
      );

      await fetchData();
    } catch (err) {
      console.error('Failed to analyze all uncalculated:', err);
    } finally {
      setIsLoading(false);
      setAnalysisProgress(null);
    }
  };

  const handleRunExperiment = async (formData: any) => {
    setIsLoading(true);
    setExperimentProgress({ current: 0, total: 100, message: 'Запуск серий экспериментов...' });
    try {
      const res = await fetch('/api/run_experiment', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(formData),
      });
      
      if (!res.ok) {
        throw new Error(await res.text() || 'Failed to run experiment');
      }

      let finalResults: any = null;
      await readProgressStream(
        res,
        (prog) => setExperimentProgress(prog),
        (results) => {
          finalResults = results;
        }
      );
      
      await fetchData();
      setIsLoading(false);
      setExperimentProgress(null);
      return finalResults;
    } catch (err) {
      setIsLoading(false);
      setExperimentProgress(null);
      throw err;
    }
  };

  return (
    <div className="min-h-screen flex flex-col bg-[#f4f7f6] text-[#2c3e50] font-sans p-4 sm:p-8">
      <nav className="flex flex-wrap gap-2 mb-6 bg-[#2c3e50] p-3 rounded-xl shadow-md font-sans">
        <button
          type="button"
          onClick={() => setActiveTab('generator')}
          className={`px-4 py-2 text-xs uppercase font-bold cursor-pointer tracking-wider rounded-lg transition-all duration-150 border-none ${
            activeTab === 'generator'
              ? 'bg-[#3498db] text-white shadow-sm'
              : 'text-gray-300 hover:bg-[#34495e] hover:text-white bg-transparent'
          }`}
        >
          1. Генератор
        </button>
        <button
          type="button"
          onClick={() => setActiveTab('analyzer')}
          className={`px-4 py-2 text-xs uppercase font-bold cursor-pointer tracking-wider rounded-lg transition-all duration-150 border-none ${
            activeTab === 'analyzer'
              ? 'bg-[#3498db] text-white shadow-sm'
              : 'text-gray-300 hover:bg-[#34495e] hover:text-white bg-transparent'
          }`}
        >

}

import React, { useRef, useEffect, useState } from 'react';

interface VisualizerProps {
  nodes: number[][] | number; // Support number of nodes or nodes array
  links: number[][];
}

export const NetworkVisualizer3D: React.FC<VisualizerProps> = ({ nodes, links }) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const [yaw, setYaw] = useState<number>(0.5);
  const [pitch, setPitch] = useState<number>(0.5);
  const [isDragging, setIsDragging] = useState<boolean>(false);
  const dragStart = useRef<{ x: number; y: number }>({ x: 0, y: 0 });

  const nodesArr = Array.isArray(nodes) ? nodes : [];

  useEffect(() => {
    setYaw(0.5);
    setPitch(0.5);
  }, [nodes]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const width = canvas.width;
    const height = canvas.height;

    ctx.fillStyle = '#0f172a';
    ctx.fillRect(0, 0, width, height);

    if (nodesArr.length === 0) return;

    const cosY = Math.cos(yaw);
    const sinY = Math.sin(yaw);
    const cosP = Math.cos(pitch);
    const sinP = Math.sin(pitch);

    const degrees = new Array(nodesArr.length).fill(0);
    links.forEach(([u, v]) => {
      if (u < degrees.length) degrees[u]++;
      if (v < degrees.length) degrees[v]++;
    });

    const projected: { x: number; y: number; z: number; originalIdx: number }[] = nodesArr.map((node, i) => {
      const cx = (node[0] ?? 0.5) - 0.5;
      const cy = (node[1] ?? 0.5) - 0.5;
      const cz = (node[2] ?? 0.5) - 0.5;

      const x1 = cx * cosY - cz * sinY;
      const z1 = cx * sinY + cz * cosY;

      const y2 = cy * cosP - z1 * sinP;
      const z2 = cy * sinP + z1 * cosP;

      const distance = 1.3;
      const factor = (width * 0.72) / (distance + z2);
      const px = x1 * factor + width / 2;
      const py = y2 * factor + height / 2;

      return { x: px, y: py, z: z2, originalIdx: i };
    });

    ctx.lineWidth = 1.0;
    links.forEach(([u, v]) => {
      if (u >= projected.length || v >= projected.length) return;
      const p1 = projected[u];
      const p2 = projected[v];

      const avgZ = (p1.z + p2.z) / 2;
      const alpha = Math.max(0.12, Math.min(0.72, 1.0 - (avgZ + 0.5)));

      const r = Math.round(((nodesArr[u]?.[0] ?? 0) + (nodesArr[v]?.[0] ?? 0)) * 110 + 35);
      const g = Math.round(((nodesArr[u]?.[1] ?? 0) + (nodesArr[v]?.[1] ?? 0)) * 110 + 100);
      const b = Math.round(((nodesArr[u]?.[2] ?? 0) + (nodesArr[v]?.[2] ?? 0)) * 110 + 145);

      ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, ${alpha})`;
      ctx.beginPath();
      ctx.moveTo(p1.x, p1.y);
      ctx.lineTo(p2.x, p2.y);
      ctx.stroke();
    });

    const sortedProjected = [...projected].sort((a, b) => b.z - a.z);

    sortedProjected.forEach((p) => {
      const origNode = nodesArr[p.originalIdx];
      if (!origNode) return;
      const deg = degrees[p.originalIdx] || 0;
      
      const r = Math.round((origNode[0] ?? 0) * 120 + 40);
      const g = Math.round((origNode[1] ?? 0) * 120 + 100);
      const b = Math.round((origNode[2] ?? 0) * 120 + 135);

      const size = Math.max(2.5, Math.min(7.0, 2.0 + Math.log2(deg + 1)));

      const grad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, size * 1.5);
      grad.addColorStop(0, `rgb(${r}, ${g}, ${b})`);
      grad.addColorStop(1, `rgba(${r}, ${g}, ${b}, 0.0)`);
      
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.arc(p.x, p.y, size * 1.5, 0, Math.PI * 2);
      ctx.fill();

      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.arc(p.x, p.y, size * 0.45, 0, Math.PI * 2);
      ctx.fill();
    });

  }, [nodesArr, links, yaw, pitch]);

  const handleMouseDown = (e: React.MouseEvent<HTMLCanvasElement>) => {
    setIsDragging(true);
    dragStart.current = { x: e.clientX, y: e.clientY };
  };

  const handleMouseMove = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (!isDragging) return;
    const dx = e.clientX - dragStart.current.x;
    const dy = e.clientY - dragStart.current.y;
    dragStart.current = { x: e.clientX, y: e.clientY };

    setYaw((prev) => prev + dx * 0.012);
    setPitch((prev) => Math.max(-Math.PI / 2.1, Math.min(Math.PI / 2.1, prev - dy * 0.012)));
  };

  const handleMouseUpOrLeave = () => {
    setIsDragging(false);
  };

  return (
    <div className="relative group rounded-xl overflow-hidden border border-slate-800 bg-slate-950 p-1 shadow-inner">
      <canvas
        ref={canvasRef}
        width={500}
        height={400}
        className="w-full h-auto cursor-grab active:cursor-grabbing block rounded-lg"
        onMouseDown={handleMouseDown}
        onMouseMove={handleMouseMove}
        onMouseUp={handleMouseUpOrLeave}
        onMouseLeave={handleMouseUpOrLeave}
      />
      <div className="absolute bottom-3 right-3 bg-slate-900/80 backdrop-blur border border-slate-700/50 px-2 py-1 rounded text-[10px] font-mono text-slate-400 select-none pointer-events-none transition opacity-75 group-hover:opacity-100">
        Перетащите для вращения в 3D
      </div>
    </div>
  );
};

        : prev.distributions.filter((d) => d !== dist);
      return { ...prev, distributions: nextDists };
    });
  };

  const computedLambda = formData.input_mode === 'c_alpha' 
    ? (formData.C / Math.pow(formData.N, formData.alpha))
    : formData.lambda_val;

  const estimateMaxNodes = (d: number, space: string, dist: string): number => {
    if (!d || d <= 0) return 200000;
    let nMax = 200000;
    if (dist.startsWith('fractal')) {
      nMax = Math.floor(0.045 / (d * d));
    } else {
      if (dist === 'gaussian') {
        nMax = Math.floor(0.125 / (d * d * d));
      } else {
        nMax = Math.floor(0.735 / (d * d * d));
      }
    }
    return Math.max(3, nMax);
  };

  const maxRecommendedNodes = estimateMaxNodes(computedLambda, formData.space_type, formData.distribution_type);

  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement>) => {
    setDuplicateWarnings([]);
    const { name, value, type } = e.target;
    let parsedValue: any = value;
    
    if (type === 'number') {
      parsedValue = parseFloat(value);
      if (isNaN(parsedValue)) parsedValue = 0;
    } else if (type === 'checkbox') {
      parsedValue = (e.target as HTMLInputElement).checked;
    }

    setFormData((prev) => ({
      ...prev,
      [name]: parsedValue,
    }));
  };

  const handleModeToggle = (mode: 'c_alpha' | 'lambda') => {
    setDuplicateWarnings([]);
    setFormData((prev) => ({
      ...prev,
      input_mode: mode,
    }));
  };

  const getPlannedJobs = () => {
    const jobs: Array<{
      N: number;
      model_type: string;
      space_type: string;
      distribution_type: string;
      er_p?: number;
      ba_m?: number;
      C?: number;
      alpha?: number;
      lambda?: number;
    }> = [];

    const spaces = formData.range_enabled 
      ? (formData.spaces.length > 0 ? formData.spaces : [formData.space_type])
      : [formData.space_type];

    const distributions = formData.range_enabled
      ? (formData.distributions.length > 0 ? formData.distributions : [formData.distribution_type])
      : [formData.distribution_type];

    if (formData.range_enabled) {
      const n_step = formData.n_step && formData.n_step > 0 ? formData.n_step : 50;
      const n_vals: number[] = [];
      let curr_n = formData.n_from;
      while (curr_n <= formData.n_to) {
        n_vals.push(curr_n);
        curr_n += n_step;
        if (n_vals.length >= 5000) break;
      }

      let p_from = 0.1;
      let p_to = 0.5;
      let p_step_val = 0.1;

      if (formData.model_type === 'lpn') {
        if (formData.input_mode === 'c_alpha') {
          p_from = formData.c_from;
          p_to = formData.c_to;
          p_step_val = formData.c_step;
        } else {
          p_from = formData.lambda_from;
          p_to = formData.lambda_to;
          p_step_val = formData.lambda_step;
        }
      } else if (formData.model_type === 'er') {
        p_from = formData.er_p_from;
        p_to = formData.er_p_to;
        p_step_val = formData.er_p_step;
      } else if (formData.model_type === 'ba') {
        p_from = formData.ba_m_from;
        p_to = formData.ba_m_to;
        p_step_val = formData.ba_m_step;
      }

      const p_step = p_step_val && p_step_val > 0 ? p_step_val : 0.01;
      const p_vals: number[] = [];
      let curr_p = p_from;
      while (curr_p <= p_to + 1e-9) {
        p_vals.push(Number(curr_p.toFixed(5)));
        curr_p += p_step;
        if (p_vals.length >= 5000) break;
      }

      if (formData.model_type === 'lpn') {
        const alpha_vals: number[] = [];
        if (formData.input_mode === 'c_alpha') {
          const a_step = formData.alpha_step && formData.alpha_step > 0 ? formData.alpha_step : 0.1;
          let curr_a = formData.alpha_from;
          while (curr_a <= formData.alpha_to + 1e-9) {
            alpha_vals.push(Number(curr_a.toFixed(5)));
            curr_a += a_step;
            if (alpha_vals.length >= 5000) break;
          }
        } else {
          alpha_vals.push(formData.alpha);
        }

        for (const sp of spaces) {
          for (const dt of distributions) {
            for (const n of n_vals) {
              for (const p of p_vals) {
                for (const al of alpha_vals) {
                  if (formData.input_mode === 'c_alpha') {
                    const computed_lam = p / Math.pow(n, al);
                    jobs.push({
                      N: n,
                      model_type: 'lpn',
                      space_type: sp,
                      distribution_type: dt,
                      C: p,
                      alpha: al,
                      lambda: computed_lam,
                    });
                  } else {
                    jobs.push({
                      N: n,
                      model_type: 'lpn',
                      space_type: sp,
                      distribution_type: dt,
                      C: formData.C,
                      alpha: al,
                      lambda: p,
                    });
                  }
                }
              }
            }
          }
        }
      } else if (formData.model_type === 'er') {
        for (const n of n_vals) {
          for (const p of p_vals) {
            jobs.push({
              N: n,
              model_type: 'er',
              space_type: 'cube',
              distribution_type: 'random',
              er_p: p,
            });
          }
        }
      } else if (formData.model_type === 'ba') {
        for (const n of n_vals) {
          for (const p of p_vals) {
            jobs.push({
              N: n,
              model_type: 'ba',
              space_type: 'cube',
              distribution_type: 'random',
              ba_m: p,
            });
          }
        }
      }
    } else {
      for (let i = 0; i < formData.count; i++) {
        if (formData.model_type === 'lpn') {
          if (formData.input_mode === 'c_alpha') {
            const computed_lam = formData.C / Math.pow(formData.N, formData.alpha);
            jobs.push({
              N: formData.N,
              model_type: 'lpn',
              space_type: formData.space_type,
              distribution_type: formData.distribution_type,
              C: formData.C,
              alpha: formData.alpha,
              lambda: computed_lam,
            });
          } else {
            jobs.push({
              N: formData.N,
              model_type: 'lpn',
              space_type: formData.space_type,
              distribution_type: formData.distribution_type,
              C: formData.C,
              alpha: formData.alpha,
              lambda: formData.lambda_val,
            });
          }
        } else if (formData.model_type === 'er') {
          jobs.push({
            N: formData.N,
            model_type: 'er',
            space_type: 'cube',
            distribution_type: 'random',
            er_p: formData.er_p,
          });
        } else if (formData.model_type === 'ba') {
          jobs.push({
            N: formData.N,
            model_type: 'ba',
            space_type: 'cube',
            distribution_type: 'random',
            ba_m: formData.ba_m,
          });
        }
      }
    }

    return jobs;
  };

  const findDuplicateNetworks = () => {
    const planned = getPlannedJobs();
    const duplicates: Network[] = [];

    for (const job of planned) {
      const found = networks.find((net) => {
        const netType = net.model_type || 'lpn';
        if (netType !== job.model_type) return false;
        if (net.N !== job.N) return false;

        if (netType === 'lpn') {
          if (net.space_type !== job.space_type) return false;
          if (net.distribution_type !== job.distribution_type) return false;

          const diffC = Math.abs((net.C ?? 0) - (job.C ?? 0));
          const diffAlpha = Math.abs((net.alpha ?? 0) - (job.alpha ?? 0));
          const diffLambda = Math.abs((net.lambda ?? 0) - (job.lambda ?? 0));

          if (diffC > 1e-4 || diffAlpha > 1e-4 || diffLambda > 1e-4) return false;
        } else if (netType === 'er') {
          if (Math.abs((net.er_p ?? 0) - (job.er_p ?? 0)) > 1e-5) return false;
        } else if (netType === 'ba') {
          if (net.ba_m !== job.ba_m) return false;
        }

        return true;
      });

      if (found && !duplicates.some((d) => d.id === found.id)) {
        duplicates.push(found);
      }
    }

    return duplicates;
  };

  const handleGenerateStream = async (skipExisting: boolean = false) => {
    setIsGenerating(true);
    setGenerationLogs(['Установка соединения с сервером...']);
    setActiveProgress(null);
    setLastGeneratedInfo(null);

    try {
      if (formData.range_enabled) {
        if (formData.spaces.length === 0) {
          throw new Error('Выберите хотя бы одну область вложения для множественной генерации.');
        }
        if (formData.distributions.length === 0) {
          throw new Error('Выберите хотя бы одно распределение узлов для множественной генерации.');
        }
      }

      const res = await fetch('/api/generate', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ ...formData, skip_existing: skipExisting }),
      });

      if (!res.ok) {
        const textLog = await res.text();
        throw new Error(textLog || 'Произошла непредвиденная ошибка на сервере');
      }

      const reader = res.body?.getReader();
      const decoder = new TextDecoder('utf-8');
      let buffer = '';

      if (reader) {
        while (true) {
          const { value, done } = await reader.read();
          if (done) break;
          buffer += decoder.decode(value, { stream: true });
          const lines = buffer.split('\n');
          buffer = lines.pop() || '';

          for (const line of lines) {
            if (line.trim()) {
              try {
                const step = JSON.parse(line);
                if (step.type === 'log') {
                  setGenerationLogs((prev) => [...prev.slice(-150), step.message]);
                  if (step.index && step.total) {
                    setActiveProgress({ current: step.index, total: step.total });
                  }
                } else if (step.type === 'success') {
                  setGenerationLogs((prev) => [...prev.slice(-150), step.message]);
                } else if (step.type === 'summary') {
                  setLastGeneratedInfo(step.message);
                  setGenerationLogs((prev) => [...prev, `🎉 ${step.message}`]);
                }
              } catch (err) {
                console.error('Error parsing streaming line:', line);
              }
            }
          }
        }
      }

      if (buffer.trim()) {
        try {
          const step = JSON.parse(buffer);
          if (step.type === 'summary') {
            setLastGeneratedInfo(step.message);
          }
        } catch (e) {}
      }

      await onGenerate(null);

    } catch (err: any) {
      setGenerationLogs((prev) => [...prev, `❌ Ошибка генерации: ${err.message}`]);
    } finally {
      setIsGenerating(false);
      setActiveProgress(null);
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    const dups = findDuplicateNetworks();
    if (dups.length > 0) {
      setDuplicateWarnings(dups);
      return;
    }
    await handleGenerateStream();
  };

  return (
    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 text-[#2c3e50] font-sans">
      <div className="lg:col-span-5 bg-white border border-[#e0e0e0] p-5 rounded-xl shadow-sm">
        <h2 className="text-sm font-bold text-[#2c3e50] uppercase tracking-wider mb-4 pb-2 border-b border-[#eee] flex items-center gap-2">
          <span>Режимы построения LPN</span>
        </h2>

        <div className="flex border-b border-gray-200 mb-4 text-xs font-bold uppercase">
          <button
            type="button"
            onClick={() => {
              setActiveTab('form');
              setDuplicateWarnings([]);
              setFormData((prev) => ({ 
                ...prev, 
                range_enabled: false,
                single_vertex_set: false,
                count: 1
              }));
            }}
            className={`pb-2 px-3 transition-colors duration-155 border-b-2 border-transparent ${
              activeTab === 'form'
                ? 'border-[#3498db] text-[#3498db]'
                : 'text-gray-500 hover:text-black hover:bg-gray-50'
            }`}
          >
            Одиночная генерация
          </button>
          <button
            type="button"
            onClick={() => {
              setActiveTab('single_p');
              setDuplicateWarnings([]);
              setFormData((prev) => ({ 
                ...prev, 
                range_enabled: false,
                single_vertex_set: true,
                count: 10
              }));
            }}
            className={`pb-2 px-3 transition-colors duration-155 border-b-2 border-transparent ${
              activeTab === 'single_p'
                ? 'border-[#3498db] text-[#3498db]'
                : 'text-gray-500 hover:text-black hover:bg-gray-50'
            }`}
          >
            Генерация для одного P
          </button>
          <button
            type="button"
            onClick={() => {
              setActiveTab('ranges');
              setDuplicateWarnings([]);
              setFormData((prev) => ({ 
                ...prev, 
                range_enabled: true 
              }));
            }}
            className={`pb-2 px-3 transition-colors duration-155 border-b-2 border-transparent ${
              activeTab === 'ranges'
                ? 'border-[#3498db] text-[#3498db]'
                : 'text-gray-500 hover:text-black hover:bg-gray-50'
            }`}
          >
            Множественная генерация
          </button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4 text-xs font-semibold">
          <div>
            <label className="block text-xs uppercase font-bold text-gray-500 mb-1 flex items-center">
              <span>Название серии сетей</span>
              <InfoTooltip text="Формирует базовое имя сети. Псевдоним 'id = n' автоматически заменяется индивидуальным идентификатором." />
            </label>
            <input
              type="text"
              name="name"
              value={formData.name}
              onChange={handleChange}
              className="w-full bg-white border border-gray-300 px-3 py-1.5 text-black font-semibold rounded-lg focus:outline-none focus:ring-2 focus:ring-[#3498db]/50 focus:border-[#3498db] transition-all shadow-sm"
            />
          </div>

          <div>
            <label className="block text-xs uppercase font-bold text-gray-700 mb-1 flex items-center">
              <span>Модель построения сети</span>
              <InfoTooltip text="Связи накладываются по правилу физических пространственных пересечений (LPN), случайным образом (ER) или по закону предпочтительного присоединения (BA)." />
            </label>
            <div className="grid grid-cols-3 gap-2 bg-gray-50 p-1 border border-gray-200 rounded-lg shadow-inner">
              <button
                type="button"
                onClick={() => setFormData((prev) => ({ 
                  ...prev, 
                  model_type: 'lpn', 
                  name: prev.name === 'Серия ER' || prev.name === 'Серия BA' || prev.name === 'Серия LPN' ? 'id = n' : prev.name 
                }))}
                className={`py-1.5 text-xs font-bold rounded-md transition-all border-none cursor-pointer ${
                  formData.model_type === 'lpn'
                    ? 'bg-[#3498db] text-white shadow-xs'
                    : 'text-gray-600 hover:bg-gray-200'
                }`}
              >
                Модель LPN
              </button>
              <button
                type="button"
                onClick={() => setFormData((prev) => ({ 
                  ...prev, 
                  model_type: 'er', 
                  name: prev.name === 'id = n' || prev.name === 'Серия LPN' || prev.name === 'Серия BA' ? 'Серия ER' : prev.name 
                }))}
                className={`py-1.5 text-xs font-bold rounded-md transition-all border-none cursor-pointer ${
                  formData.model_type === 'er'
                    ? 'bg-[#3498db] text-white shadow-xs'
                    : 'text-gray-600 hover:bg-gray-200'
                }`}
              >
                Эрдеша-Реньи (ER)
              </button>
              <button
                type="button"
                onClick={() => setFormData((prev) => ({ 
                  ...prev, 
                  model_type: 'ba', 
                  name: prev.name === 'id = n' || prev.name === 'Серия LPN' || prev.name === 'Серия ER' ? 'Серия BA' : prev.name 
                }))}
                className={`py-1.5 text-xs font-bold rounded-md transition-all border-none cursor-pointer ${
                  formData.model_type === 'ba'
                    ? 'bg-[#3498db] text-white shadow-xs'
                    : 'text-gray-600 hover:bg-gray-200'
                }`}
              >
                Барабаши-Альберта (BA)
              </button>
            </div>
          </div>

          {(activeTab === 'form' || activeTab === 'single_p') ? (
            <>
              <div>
                <label className="block text-xs uppercase font-bold text-gray-700 mb-1 flex items-center">
                  <span>Кол-во вершин (N)</span>
                  <InfoTooltip text="Количество генерируемых центров сфер в выбранной ограниченной 3D локализации." />
                </label>
                <input
                  type="text"
                  name="N"
                  value={stringInputs.N ?? formData.N ?? ''}
                  onFocus={(e) => e.target.select()}
                  onChange={handleStringChange}
                  className="w-full bg-white border border-gray-300 px-3 py-1.5 text-black font-mono font-bold rounded-lg focus:outline-none focus:ring-2 focus:ring-[#3498db]/50 focus:border-[#3498db] transition-all shadow-sm"
                />
                
                {formData.model_type === 'lpn' && (
                  <div className="mt-2 p-2.5 rounded-lg border text-[11px] leading-relaxed bg-gray-50/50 border-gray-200">
                    <div className="flex justify-between items-center mb-1 text-gray-700 font-mono font-bold">
                      <span className="font-semibold text-[10px] uppercase text-gray-500">Лимит вместимости:</span>
                      <span className="font-bold text-gray-900">
                        {formData.nodes_as_points ? 'бесконечность' : `${maxRecommendedNodes === Infinity ? 'бесконечность' : `${maxRecommendedNodes} вершин`}`}
                      </span>
                    </div>
                    {(!formData.nodes_as_points && maxRecommendedNodes !== Infinity && formData.N > maxRecommendedNodes) ? (
                      <div className="text-amber-700 bg-amber-50 border-amber-100 border px-2 py-0.5 rounded font-bold text-[10px]">
                        ⚠️ Возможен длительный расчет (N больше лимита вместимости).
                      </div>
                    ) : (
                      <div className="text-emerald-700 bg-emerald-50 border-emerald-100 border px-2 py-0.5 rounded font-bold text-[10px]">
                        ✓ Допустимое кол-во вершин.
                      </div>
                    )}
                  </div>
                )}
              </div>

              {formData.model_type === 'lpn' && (
                <div>
                  <label className="block text-xs uppercase font-bold text-gray-700 mb-1">
                    Режим задания параметров физичности
                  </label>
                  <div className="grid grid-cols-2 gap-2 bg-gray-50 p-1 border border-gray-200 rounded-lg shadow-inner">
                    <button
                      type="button"
                      onClick={() => handleModeToggle('lambda')}
                      className={`py-1.5 text-xs font-bold rounded-md transition-all border-none cursor-pointer ${
                        formData.input_mode === 'lambda'
                          ? 'bg-[#3498db] text-white shadow-xs'
                          : 'text-gray-600 hover:bg-gray-200'
                      }`}
                    >
                      Параметр λ
                    </button>
                    <button
                      type="button"
                      onClick={() => handleModeToggle('c_alpha')}
                      className={`py-1.5 text-xs font-bold rounded-md transition-all border-none cursor-pointer ${
                        formData.input_mode === 'c_alpha'
                          ? 'bg-[#3498db] text-white shadow-xs'
                          : 'text-gray-650 hover:bg-gray-200'
                      }`}
                    >
                      Параметры C + α
                    </button>
                  </div>
                </div>
              )}

              {formData.model_type === 'lpn' && (
                formData.input_mode === 'c_alpha' ? (
                  <div className="grid grid-cols-2 gap-3 bg-gray-50 p-2.5 border border-dashed border-gray-300 rounded-lg">
                    <div>
                      <label className="block text-[10px] uppercase font-bold text-gray-600 mb-1">
                        Константа С
                      </label>

                        type="text"
                        name="C"
                        value={stringInputs.C ?? formData.C ?? ''}
                        onFocus={(e) => e.target.select()}
                        onChange={handleStringChange}
                        className="w-full bg-white border border-gray-300 rounded-lg px-2 py-1 text-black font-mono font-bold focus:outline-none focus:ring-2 focus:ring-[#3498db]/50"
                      />
                    </div>
                    <div>
                      <label className="block text-[10px] uppercase font-bold text-gray-600 mb-1">
                        α
                      </label>
                      <input
                        type="text"
                        name="alpha"
                        value={stringInputs.alpha ?? formData.alpha ?? ''}
                        onFocus={(e) => e.target.select()}
                        onChange={handleStringChange}
                        className="w-full bg-white border border-gray-300 rounded-lg px-2 py-1 text-black font-mono font-bold focus:outline-none focus:ring-2 focus:ring-[#3498db]/50"
                      />
                    </div>
                  </div>
                ) : (
                  <div className="bg-gray-50 p-2.5 border border-dashed border-gray-300 rounded-lg">
                    <label className="block text-xs uppercase font-bold text-gray-600 mb-1">
                      Ширина λ
                    </label>
                    <input
                      type="text"
                      name="lambda_val"
                      value={stringInputs.lambda_val ?? formData.lambda_val ?? ''}
                      onFocus={(e) => e.target.select()}
                      onChange={handleStringChange}
                      className="w-full bg-white border border-gray-300 rounded-lg px-3 py-1.5 text-black font-mono font-bold focus:outline-none focus:ring-2 focus:ring-[#3498db]/50"
                    />
                  </div>
                )
              )}

              {formData.model_type === 'er' && (
                <div className="bg-gray-50 p-2.5 border border-dashed border-gray-300 rounded-lg">
                  <label className="block text-xs uppercase font-bold text-gray-700 mb-1">
                    Вероятность связи (p)
                  </label>
                  <input
                    type="text"
                    name="er_p"
                    value={stringInputs.er_p ?? formData.er_p ?? ''}
                    onFocus={(e) => e.target.select()}
                    onChange={handleStringChange}
                    className="w-full bg-white border border-gray-300 rounded-lg px-3 py-1.5 text-black font-mono font-bold focus:outline-none focus:ring-2 focus:ring-[#3498db]/50"
                  />
                  <span className="block text-[10px] text-gray-550 mt-1 leading-tight font-bold">
                    Каждая из N*(N-1)/2 возможных связей создается независимо с вероятностью p.
                  </span>
                </div>
              )}

              {formData.model_type === 'ba' && (
                <div className="bg-gray-50 p-2.5 border border-dashed border-gray-300 rounded-lg">
                  <label className="block text-xs uppercase font-bold text-gray-700 mb-1">
                    Число связей на новый узел (m)
                  </label>
                  <input
                    type="text"
                    name="ba_m"
                    value={stringInputs.ba_m ?? formData.ba_m ?? ''}
                    onFocus={(e) => e.target.select()}
                    onChange={handleStringChange}
                    className="w-full bg-white border border-gray-300 rounded-lg px-3 py-1.5 text-black font-mono font-bold focus:outline-none focus:ring-2 focus:ring-[#3498db]/50"
                  />
                  <span className="block text-[10px] text-gray-550 mt-1 leading-tight font-bold">
                    При добавлении очередного узла он подсоединяется к m существующим узлам по принципу предпочтительного присоединения.
                  </span>
                </div>

                  Длина ребра d_avg: <strong className="font-mono text-amber-900">{hoveredPoint.lenVal.toFixed(6)}</strong>
                </div>
                <div className="bg-emerald-50 text-emerald-800 border border-emerald-100 px-2 py-0.5 rounded-md">
                  M_phys ср: <strong className="font-mono text-emerald-900">{hoveredPoint.m_phys_avg.toFixed(1)}</strong>
                </div>
                <div className="bg-indigo-50 text-indigo-800 border border-indigo-100 px-2 py-0.5 rounded-md">
                  M_max ср: <strong className="font-mono text-indigo-900">{hoveredPoint.m_max_avg.toFixed(1)}</strong>
                </div>
              </div>
            </div>
          ) : hoveredLineResult ? (
            <div className="flex items-center justify-between gap-2">
              <div className="flex items-center gap-2">
                {(() => {
                  const curveKey = `${hoveredLineResult.experiment_id}_${hoveredLineResult.space_type}_${hoveredLineResult.distribution_type}_${hoveredLineResult.lambda}`;
                  const idx = resultsList.indexOf(hoveredLineResult);
                  const currentColor = customColors[curveKey] || colors[idx % colors.length];
                  return <span className="h-2.5 w-2.5 rounded shadow-sm" style={{ backgroundColor: currentColor }} />;
                })()}
                <span className="font-bold text-gray-800">{getSpaceRuLocal(hoveredLineResult.space_type)} ({getDistRuLocal(hoveredLineResult.distribution_type)})</span>
                <span className="text-gray-400 font-mono">λ = {hoveredLineResult.lambda.toFixed(5)}</span>
              </div>
              <div className="flex items-center gap-4 text-[11px] font-mono">
                <div>
                  <span className="text-gray-500">M_phys (ср):</span> <strong className="text-emerald-700">{hoveredLineResult.m_phys_avg.toFixed(2)}</strong>
                </div>
                <div>
                  <span className="text-gray-500">M_max (ср):</span> <strong className="text-indigo-700">{hoveredLineResult.m_max_avg.toFixed(2)}</strong>
                </div>
              </div>
            </div>
          ) : (
            <p className="text-gray-400 italic text-center text-[10px] py-1.5 uppercase tracking-wide font-bold">
              Наведите курсор на графическую линию или отдельную точку для получения подробных численных координат.
            </p>
          )}
        </div>
      </div>
    );
  };

  const renderHistogram = (resultsList: any[], field: 'm_phys_list' | 'm_max_list', titleStr: string) => {
    let allValues: number[] = [];
    resultsList.forEach((r) => {
      if (r[field]) {
        allValues.push(...r[field]);
      }
    });

    if (allValues.length === 0) {
      allValues = [0, 100];
    }

    const minVal = Math.min(...allValues);
    const maxVal = Math.max(...allValues);
    const range = Math.max(1, maxVal - minVal);
    const valDiff = maxVal - minVal;
    
    const isAllIntegers = allValues.every((v) => Number.isInteger(v));
    const isSmallIntRange = isAllIntegers && valDiff >= 0 && valDiff <= 15;
    const binCount = isSmallIntRange ? Math.round(valDiff) + 1 : 12;
    const binWidth = isSmallIntRange ? 1 : (valDiff === 0 ? 1 : valDiff / binCount);

    const binnedSeries = resultsList.map((r, rIdx) => {
      const bins = new Array(binCount).fill(0);
      const values = r[field] || [];
      values.forEach((v: number) => {
        let bIdx = Math.floor((v - minVal) / binWidth);
        if (bIdx >= binCount) bIdx = binCount - 1;
        if (bIdx < 0) bIdx = 0;
        bins[bIdx]++;
      });

      const curveKey = `${r.experiment_id}_${r.space_type}_${r.distribution_type}_${r.lambda}`;
      const defaultColor = colors[rIdx % colors.length];
      const currentColor = customColors[curveKey] || defaultColor;

      return {
        rIdx,
        bins,
        color: currentColor,
        curveKey,
        label: r.label,
      };
    });

    let maxFreq = 0;
    binnedSeries.forEach((s) => {
      const peak = Math.max(...s.bins);
      if (peak > maxFreq) maxFreq = peak;
    });
    if (maxFreq === 0) maxFreq = 1;

    const width = 500;
    const height = 180;
    const padding = 35;

    const scaleX = (binIdx: number) => {
      return padding + (binIdx / binCount) * (width - padding * 2);
    };

    const scaleY = (freq: number) => {
      return height - padding - (freq / maxFreq) * (height - padding * 2);
    };

    const renderBinXLabel = (binIdx: number) => {
      const val = minVal + binIdx * binWidth;
      if (isSmallIntRange) {
        return val.toFixed(0);
      }
      if (range < 5) {
        return val.toFixed(2);
      }
      if (range < 50) {
        return val.toFixed(1);
      }
      return val >= 1000 ? `${(val/1000).toFixed(1)}k` : val.toFixed(0);
    };

    return (
      <div className="relative bg-white p-4 border border-gray-200 rounded-lg shadow-sm font-bold">
        <h4 className="text-xs font-bold text-[#2c3e50] uppercase mb-2">{titleStr}</h4>
        
        <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto">
          <line x1={padding} y1={height - padding} x2={width - padding} y2={height - padding} stroke="#bdc3c7" strokeWidth={1} />
          <line x1={padding} y1={padding} x2={padding} y2={height - padding} stroke="#bdc3c7" strokeWidth={1} />
          
          {[0, 0.5, 1.0].map((term) => {
            const freq = Math.round(term * maxFreq);
            const y = scaleY(freq);
            return (
              <g key={term}>
                <line x1={padding} y1={y} x2={width - padding} y2={y} stroke="#f0f3f4" strokeWidth={0.5} strokeDasharray="2,2" />
                <text x={padding - 6} y={y + 3} textAnchor="end" className="text-[8px] fill-gray-500 font-mono font-semibold">
                  {freq}
                </text>
              </g>
            );
          })}

          {Array.from({ length: binCount }).map((_, bIdx) => {
            const xStart = scaleX(bIdx);
            const xEnd = scaleX(bIdx + 1);
            const wOuter = xEnd - xStart;
            
            return binnedSeries.map((s, idx) => {
              const freq = s.bins[bIdx];
              if (freq === 0) return null;
              
              const hValue = (freq / maxFreq) * (height - padding * 2);
              const yStart = height - padding - hValue;
              
              const seriesCount = binnedSeries.length;
              const barWidth = wOuter / seriesCount * 0.85;
              const xShifted = xStart + idx * (wOuter / seriesCount) + wOuter/seriesCount*0.075;

              const isBarHighlighted = hoveredLine === s.curveKey;
              const isSomeLineHighlighted = hoveredLine !== null;

              const labelClean = s.label?.split('[')[0] || '';
              const binRangeStr = isSmallIntRange
                ? `M = ${minVal + bIdx}`
                : `Интервал [${renderBinXLabel(bIdx)} - ${renderBinXLabel(bIdx + 1)}]`;

              const isBarHovered = hoveredBar && 
                hoveredBar.field === field && 
                hoveredBar.curveKey === s.curveKey && 
                hoveredBar.bIdx === bIdx;

              return (
                <rect
                  key={s.rIdx}
                  x={xShifted}
                  y={yStart}
                  width={barWidth}
                  height={hValue}
                  fill={s.color}
                  fillOpacity={isBarHovered ? 1.0 : (isSomeLineHighlighted ? (isBarHighlighted ? 0.9 : 0.15) : 0.65)}
                  stroke={isBarHovered || isBarHighlighted ? "#2c3e50" : "#7f8c8d"}
                  strokeWidth={isBarHovered ? 1.5 : (isBarHighlighted ? 1 : 0.5)}
                  className="transition duration-100 cursor-help"
                  onMouseEnter={() => {
                    setHoveredLine(s.curveKey);
                    setHoveredBar({
                      field,
                      curveKey: s.curveKey,
                      bIdx,
                      freq,
                      x: xShifted + barWidth / 2,
                      y: yStart,
                      color: s.color,
                      label: labelClean,
                      binRangeStr,
                    });
                  }}
                  onMouseLeave={() => {
                    setHoveredLine(null);
                    setHoveredBar(null);
                  }}
                />
              );
            });
          })}

          {Array.from({ length: binCount + 1 }).map((_, bIdx) => {
            if (binCount > 8 && bIdx % 2 !== 0 && bIdx !== binCount) return null;
            const x = scaleX(bIdx);
            return (
              <text key={bIdx} x={x} y={height - padding + 13} textAnchor="middle" className="text-[8px] fill-gray-500 font-mono font-semibold">
                {renderBinXLabel(bIdx)}
              </text>
            );
          })}
        </svg>

        {hoveredBar && hoveredBar.field === field && (
          <div 
            className="absolute z-35 pointer-events-none bg-slate-900/95 backdrop-blur text-white border border-slate-700/80 p-2.5 rounded-lg shadow-xl text-[10px] leading-relaxed max-w-[210px]"
            style={{
              left: `${(hoveredBar.x / width) * 100}%`,
              top: `${(hoveredBar.y / height) * 100}%`,
              transform: 'translate(-50%, -100%) translateY(-8px)',
            }}
          >
            <div className="absolute top-full left-1/2 -translate-x-1/2 border-4 border-transparent border-t-slate-900/95" />
            
            <div className="flex items-center gap-1.5 font-bold mb-1 pb-1 border-b border-slate-800 text-slate-200">
              <span className="h-2 w-2 rounded-full inline-block flex-shrink-0" style={{ backgroundColor: hoveredBar.color }} />
              <span className="truncate">{hoveredBar.label}</span>
            </div>
            
            <div className="space-y-1 font-mono">
              <div className="text-amber-400 font-bold">
                {hoveredBar.binRangeStr}
              </div>
              <div className="flex justify-between gap-3 pt-0.5 border-t border-slate-800/50">
                <span className="text-slate-400">Частота:</span>
                <span className="text-sky-300 font-bold">
                  {hoveredBar.freq} {hoveredBar.freq === 1 ? 'повтор' : (hoveredBar.freq > 1 && hoveredBar.freq < 5 ? 'повтора' : 'повторов')}
                </span>
              </div>
            </div>
          </div>
        )}
      </div>
    );
  };

  return (
    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 text-[#2c3e50] font-sans">
      <div className="lg:col-span-4 bg-white border border-[#e0e0e0] p-5 rounded-xl shadow-sm">
        <h2 className="text-sm font-bold text-[#2c3e50] uppercase tracking-wider mb-4 pb-2 border-b border-[#eee] flex items-center gap-1.5">
          <span>Параметры симуляции</span>
          <InfoTooltip text="Настройки масштабного статистического прогона экспериментов средней длины ребер на различных моделях (LPN, ER, BA) в пространстве." />
        </h2>

        <form onSubmit={handleSubmit} className="space-y-4 text-xs font-bold">
          <div className="bg-gray-50 border border-gray-200 p-3 rounded-lg leading-relaxed shadow-inner">
            <label className="block text-[10px] font-bold text-gray-500 uppercase mb-1">
              Название эксперимента (Формируется автоматически)
            </label>
            <div className="text-xs font-bold text-[#3498db] break-words">
              {`N=${formData.N}, λ=[${formData.lambdasString.split(',').map(x => x.trim()).filter(x => x.length > 0).join('; ')}], Модели: ${formData.model_types.map(m => m.toUpperCase()).join('+')}, Область: ${formData.spaces.map(s => s === 'cube' ? 'Куб' : s === 'sphere' ? 'Шар' : s).join('+')}, Распределение: ${formData.distributions.map(d => d === 'random' ? 'Равномерное' : d === 'gaussian' ? 'Гаусс' : d === 'grid' ? 'Сетка' : d === 'fractal_tetrahedron' ? 'Тетраэдр (в кубе)' : d === 'fractal_tetrahedron_cropped' ? 'Тетраэдр (вырезан)' : d === 'fractal_octahedron' ? 'Октаэдр (в кубе)' : d === 'fractal_octahedron_cropped' ? 'Октаэдр (вырезан)' : d === 'fractal_menger' ? 'Губка' : d === 'fractal_cube' ? 'Кантор' : d).join('+')}`}
            </div>
          </div>

          <div>
            <label className="block text-xs font-bold text-gray-500 uppercase mb-1 flex items-center">
              <span>Количество узлов сети N</span>
              <InfoTooltip text="Количество генерируемых центров сфер (узлов сетей) на один прогон." />
            </label>
            <input
              type="text"
              name="N"
              value={stringInputs.N ?? formData.N ?? ''}
              onChange={handleStringChange}
              className="w-full bg-white border border-gray-300 px-3 py-1.5 text-black font-mono font-bold rounded-lg focus:outline-none focus:ring-2 focus:ring-[#3498db]/50 focus:border-[#3498db] shadow-sm transition-all"
            />
          </div>

          <div>
            <label className="block text-xs font-bold text-gray-500 uppercase mb-1 border-b border-gray-200 pb-0.5 flex items-center">
              <span>Варьируемые значения λ (через запятую)</span>
              <InfoTooltip text="Шаг варьирования параметра исключения. Будет построена отдельная статистическая кривая усадки для каждого значения." />
            </label>
            <input
              type="text"
              name="lambdasString"
              value={formData.lambdasString}
              onChange={handleInputChange}
              className="w-full bg-white border border-gray-300 px-3 py-1.5 text-black font-mono text-xs font-bold rounded-lg focus:outline-none focus:ring-2 focus:ring-[#3498db]/50 focus:border-[#3498db] shadow-sm transition-all"
              placeholder="0.0001, 0.05, 0.1"
            />
            <p className="text-[10px] text-gray-450 mt-1 italic leading-tight">
              Будет построена отдельная кривая усреднения для каждого аргумента Lambda.
            </p>
          </div>

          <div>
            <label className="block text-xs font-bold text-gray-500 uppercase mb-1">
              Число прогонов для усреднения
            </label>
            <input
              type="text"
              name="repetitions"
              value={stringInputs.repetitions ?? formData.repetitions ?? ''}
              onChange={handleStringChange}
              className="w-full bg-white border border-gray-300 px-3 py-1.5 text-black font-mono font-bold rounded-lg focus:outline-none focus:ring-2 focus:ring-[#3498db]/50 focus:border-[#3498db] shadow-sm transition-all"
            />
            <p className="text-[10px] text-gray-450 mt-1 italic leading-snug">Больше повторов — более точное распределение в гистограммах.</p>
          </div>

          <div className="flex items-center gap-2 pt-1 pb-2">
            <input
              type="checkbox"
              id="exp_nodes_as_points"
              checked={formData.nodes_as_points}
              onChange={() => setFormData(prev => ({ ...prev, nodes_as_points: !prev.nodes_as_points }))}
              className="rounded text-[#3498db] border-gray-300 focus:ring-0 h-4 w-4 cursor-pointer"
            />
            <label htmlFor="exp_nodes_as_points" className="text-xs font-bold text-gray-650 cursor-pointer select-none">
              Вершины как точки (без диаметра)
            </label>
          </div>

          <div className="bg-gray-50 p-3.5 border border-dashed border-gray-300 rounded-lg">
            <label className="block text-xs font-bold text-gray-500 uppercase mb-2">Генерация сетевых моделей</label>
            <div className="flex flex-col gap-2 font-bold text-gray-700">
              <label className="flex items-center gap-1.5 text-xs cursor-pointer select-none">
                <input
                  type="checkbox"
                  checked={formData.model_types.includes('lpn')}
                  onChange={() => handleModelTypeToggle('lpn')}
                   className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                LPN (Линейно-физическая сеть)
              </label>
              <label className="flex items-center gap-1.5 text-xs cursor-pointer select-none">
                <input
                  type="checkbox"
                  checked={formData.model_types.includes('er')}
                  onChange={() => handleModelTypeToggle('er')}
                   className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                ER (Случайная Эрдёша-Реньи)
              </label>
              <label className="flex items-center gap-1.5 text-xs cursor-pointer select-none">
                <input
                  type="checkbox"
                  checked={formData.model_types.includes('ba')}
                  onChange={() => handleModelTypeToggle('ba')}
                   className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                BA (Безмасштабная Барабаши-Альберт)
              </label>
            </div>
          </div>

          <div className="bg-gray-50 p-3.5 border border-dashed border-gray-300 rounded-lg">
            <label className="block text-xs font-bold text-gray-500 uppercase mb-2">Область локализации</label>
            <div className="flex flex-col gap-2">
              <label className="flex items-center gap-1.5 text-xs text-gray-700 cursor-pointer select-none font-bold">
                <input
                  type="checkbox"
                  checked={formData.spaces.includes('cube')}
                  onChange={() => handleSpaceToggle('cube')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                3D Куб
              </label>
              <label className="flex items-center gap-1.5 text-xs text-gray-700 cursor-pointer select-none font-bold">
                <input
                  type="checkbox"
                  checked={formData.spaces.includes('sphere')}
                  onChange={() => handleSpaceToggle('sphere')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Шар
              </label>
            </div>
          </div>

          <div className="bg-gray-50 p-3.5 border border-dashed border-gray-300 rounded-lg">
            <label className="block text-xs font-bold text-gray-500 uppercase mb-2">Распределение вершин</label>
            <div className="space-y-1.5 text-xs text-gray-700 font-bold">
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('random')}
                  onChange={() => handleDistToggle('random')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Случайное
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('gaussian')}
                  onChange={() => handleDistToggle('gaussian')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Нормальное (Гаусс)
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('grid')}
                  onChange={() => handleDistToggle('grid')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Периодическая решетка
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('fractal_tetrahedron')}
                  onChange={() => handleDistToggle('fractal_tetrahedron')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Тетраэдр Серпинского (в кубе)
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('fractal_tetrahedron_cropped')}
                  onChange={() => handleDistToggle('fractal_tetrahedron_cropped')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Тетраэдр Серпинского (вырезан)
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('fractal_octahedron')}
                  onChange={() => handleDistToggle('fractal_octahedron')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Октаэдр Серпинского (в кубе)
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('fractal_octahedron_cropped')}
                  onChange={() => handleDistToggle('fractal_octahedron_cropped')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Октаэдр Серпинского (вырезан)
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('fractal_menger')}
                  onChange={() => handleDistToggle('fractal_menger')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Губка Менгера
              </label>
              <label className="flex items-center gap-1.5 cursor-pointer">
                <input
                  type="checkbox"
                  checked={formData.distributions.includes('fractal_cube')}
                  onChange={() => handleDistToggle('fractal_cube')}
                  className="rounded text-[#3498db] border-gray-300 focus:ring-0 w-4 h-4 cursor-pointer"
                />
                Канторова пыль
              </label>
            </div>
          </div>

          <div className="pt-2">
            <button
              type="submit"
              disabled={isLoading}
              className="w-full bg-[#3498db] hover:bg-[#2980b9] disabled:bg-gray-300 disabled:text-gray-500 text-white font-bold py-2.5 px-4 transition rounded-lg shadow-sm flex items-center justify-center gap-2 cursor-pointer uppercase tracking-wider text-xs border-0"
            >
              {isLoading ? (
                <>
                  <RotateCw className="animate-spin h-3.5 w-3.5" />
                  Вычисление... ({formData.repetitions} повтор.)
                </>
              ) : (
                <>
                  <Play className="h-3.5 w-3.5" />
                  Запустить расчеты
                </>
              )}
            </button>

            {progress && (
              <div className="mt-4 p-3 bg-indigo-50 border border-indigo-150 rounded-lg shadow-inner">
                <div className="flex justify-between items-center text-[10px] font-bold text-indigo-700 mb-1">
                  <span>Прогресс эксперимента:</span>
                  <span>{progress.current} / {progress.total || 100} ({Math.round((progress.current / (progress.total || 100)) * 100)}%)</span>
                </div>
                <div className="w-full bg-indigo-100 rounded-full h-1.5 overflow-hidden">
                  <div 
                    className="bg-indigo-600 h-1.5 rounded-full transition-all duration-300"
                    style={{ width: `${(progress.current / (progress.total || 100)) * 100}%` }}
                  />
                </div>
                <p className="text-[9px] text-indigo-500 mt-1 italic font-mono truncate" title={progress.message}>
                  {progress.message}
                </p>
              </div>
            )}
          </div>
        </form>

        <button
          type="button"
          onClick={() => setShowHistory(!showHistory)}
          className="w-full mt-4 bg-gray-100 hover:bg-gray-200 border border-gray-200 text-gray-700 font-bold py-2 rounded-lg transition text-xs flex items-center justify-center gap-1.5 cursor-pointer uppercase tracking-tight"
        >
          <History className="h-3.5 w-3.5" />
          {showHistory ? 'Скрыть архив экспериментов' : 'Открыть архив экспериментов'}
        </button>

                        </span>

                      <div className="flex justify-between items-center text-[9px] font-mono text-gray-400 font-semibold pl-4 font-bold">
                        <span>Сетей в БД: {matched.length}</span>
                        {loadingCount > 0 ? (
                          <span className="text-blue-500 animate-pulse">Загрузка {loadedCount}/{matched.length}...</span>
                        ) : (
                          <span className="text-gray-500 text-[10px] flex items-center gap-0.5">
                            <Check className="h-3 w-3 text-green-500" /> Готово
                          </span>
                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            )}
          </div>

          {isSomeLoading && (
            <div className="mt-3 flex items-center font-mono font-bold text-[10px] text-blue-500 gap-1 bg-blue-50 border border-blue-100 p-2 rounded-lg shrink-0">
              <RefreshCw className="animate-spin h-3.5 w-3.5" />
              <span>Фоновая подгрузка реберных массивов из SQLite...</span>
            </div>
          )}
        </div>

        <div className="lg:col-span-8 flex flex-col items-center justify-center">
          {enabledProfiles.length === 0 ? (
            <div className="flex flex-col items-center justify-center p-12 text-gray-400 italic text-xs h-64 border border-dashed border-gray-200 rounded-xl bg-gray-50/50 w-full col-span-3">
              Включите хотя бы один настроенный профиль в меню управления выше.
            </div>
          ) : activeNetworks.length === 0 ? (
            <div className="flex flex-col items-center justify-center p-12 text-gray-400 italic text-xs h-64 border border-dashed border-gray-200 rounded-xl bg-gray-50/50 w-full col-span-3">
              Нет сетей в базе данных под критерии выбранных профилей. Сгенерируйте их во вкладке ГЕНЕРАЦИЯ.
            </div>
          ) : curves.length === 0 ? (
            <div className="flex flex-col items-center justify-center p-12 text-blue-500 text-xs h-64 border border-dashed border-gray-200 rounded-xl bg-gray-50/50 w-full font-mono font-bold col-span-3">
              <RefreshCw className="animate-spin h-6 w-6 mb-2 text-blue-500" /> Загрузка графов и весов степеней из SQLite...
            </div>
          ) : (
            <div className="w-full relative">
              <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto overflow-visible select-none">
                <defs>
                  <clipPath id="pk-compare-clip">
                    <rect x={paddingLeft} y={paddingTop} width={chartWidth} height={chartHeight} />
                  </clipPath>
                </defs>

                {yTicks.map((yVal, idx) => {
                  const y = getY(yVal);
                  return (
                    <g key={idx}>
                      <line
                        x1={paddingLeft}
                        y1={y}
                        x2={width - paddingRight}
                        y2={y}
                        stroke="#f0f0f0"
                        strokeWidth="1"
                      />
                      <text
                        x={paddingLeft - 8}
                        y={y + 3}
                        textAnchor="end"
                        className="fill-gray-400 font-mono text-[9px] font-bold"
                      >
                        {(yVal * 100).toFixed(0)}%
                      </text>
                    </g>
                  );
                })}

                {xTicks.map((kVal, idx) => {
                  const x = getX(kVal);
                  return (
                    <g key={idx}>
                      <line
                        x1={x}
                        y1={paddingTop}
                        x2={x}
                        y2={paddingTop + chartHeight}
                        stroke="#f5f5f5"
                        strokeWidth="1"
                      />
                      <text
                        x={x}
                        y={paddingTop + chartHeight + 14}
                        textAnchor="middle"
                        className="fill-gray-400 font-mono text-[9px] font-bold"
                      >
                        k={kVal}
                      </text>
                    </g>
                  );
                })}

                <line
                  x1={paddingLeft}
                  y1={paddingTop + chartHeight}
                  x2={width - paddingRight}
                  y2={paddingTop + chartHeight}
                  stroke="#ccc"
                  strokeWidth="1.5"
                />
                <line
                  x1={paddingLeft}
                  y1={paddingTop}
                  x2={paddingLeft}
                  y2={paddingTop + chartHeight}
                  stroke="#ccc"
                  strokeWidth="1.5"
                />

                <g clipPath="url(#pk-compare-clip)">
                  {curves.map((curve) => {
                    const pathD = makePathD(curve.distribution);
                    const areaD = makeAreaPathD(curve.distribution);

                    return (
                      <g key={curve.id}>
                        <path
                          d={areaD}
                          fill={curve.color}
                          fillOpacity={displayMode === 'profiles' ? 0.15 : 0.05}
                          className="transition-all duration-300"
                        />
                        <path
                          d={pathD}
                          fill="none"
                          stroke={curve.color}
                          strokeWidth={curve.isProfileAverage ? 3 : 1.8}
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          strokeDasharray={curve.isProfileAverage ? "0" : "5,3"}
                          className="transition-all duration-300"
                        />
                        {curve.distribution.map((item, pIdx) => {
                          const cx = getX(item.k);
                          const cy = getY(item.p);
                          const isKHovered = hoveredK === item.k;
                          return (
                            <circle
                              key={`${curve.id}-dot-${pIdx}`}
                              cx={cx}
                              cy={cy}
                              r={isKHovered ? 5 : curve.isProfileAverage ? 3.5 : 2.5}
                              fill={curve.color}
                              stroke="#fff"
                              strokeWidth={isKHovered ? 2 : 1}
                              onMouseEnter={() => setHoveredK(item.k)}
                              onMouseLeave={() => setHoveredK(null)}
                              className="cursor-pointer transition-all duration-150"
                            />
                          );
                        })}
                      </g>
                    );
                  })}
                </g>
              </svg>

              <div className="h-10 mt-2.5 flex justify-center items-center font-mono text-[9px] text-gray-600 bg-gray-50 border border-gray-150 rounded-lg px-3">
                {hoveredK !== null ? (
                  <div className="flex flex-wrap items-center justify-center gap-3">
                    <span>Степень <strong className="text-gray-950 font-bold text-[10px]">k = {hoveredK}</strong></span>
                    <span className="text-gray-300">|</span>
                    <div className="flex flex-wrap gap-x-3 gap-y-1 items-center">
                      {curves.slice(0, 5).map((curve) => {
                        const prob = curve.distribution.find((d) => d.k === hoveredK)?.p || 0;
                        return (
                          <span key={curve.id} style={{ color: curve.color }} className="font-bold">
                            {curve.name}: {(prob * 100).toFixed(1)}%
                          </span>
                        );
                      })}
                      {curves.length > 5 && <span className="text-gray-400 italic">и ещё {curves.length - 5} сетей</span>}
                    </div>
                  </div>
                ) : (
                  <span className="text-gray-400 italic text-center leading-tight">
                    Наведите курсор мыши на маркеры на графике, чтобы в реальном времени считать точную удельную долю вершин P(k) при выбранной степени k.
                  </span>
                )}
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};
